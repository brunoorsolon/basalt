# Native Inkling 2 engine from Project Maya. The host expert kernels use x86 intrinsics.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "strata-ik2 requires Linux x86-64")
endif()

if(STRATA_ENABLE_HIP)
  if(NOT STRATA_HIP_ARCHS STREQUAL "gfx1201")
    message(FATAL_ERROR "strata-ik2 HIP candidates target only one R9700 (gfx1201)")
  endif()
  # Source LANGUAGE properties are directory-scoped; the root HIP glob does not reach this subdirectory.
  set_source_files_properties(ik2_kernels.cu ik2_main.cu ik2_draft.cu ../../tests/ik2_kernels_test.cu
    PROPERTIES LANGUAGE HIP)
else()
set(ik2_archs)
foreach(arch IN LISTS CMAKE_CUDA_ARCHITECTURES)
  if(arch MATCHES "^90(-real|-virtual)?$")
    list(APPEND ik2_archs "90a${CMAKE_MATCH_1}")
  elseif(arch MATCHES "^([0-9]+)a?(-real|-virtual)?$")
    if(CMAKE_MATCH_1 GREATER_EQUAL 80)
      list(APPEND ik2_archs "${arch}")
    endif()
  else()
    message(FATAL_ERROR "For strata-ik2, specify numeric CUDA architectures (e.g. 86)")
  endif()
endforeach()
if(NOT ik2_archs)
  message(FATAL_ERROR "strata-ik2 requires an Ampere or newer GPU (compute capability 8.0+)")
endif()
set(CMAKE_CUDA_ARCHITECTURES ${ik2_archs})
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()

find_package(Threads REQUIRED)
add_library(ik2_kernels STATIC ik2_kernels.cu)
target_link_libraries(ik2_kernels PUBLIC ${_strata_gpu_runtime_target})
add_executable(strata-ik2 ik2_main.cu ik2_draft.cu ik2_cpu.cpp)
target_include_directories(strata-ik2 PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(strata-ik2 PRIVATE ik2_kernels ${_strata_gpu_blas_target} Threads::Threads)

find_package(OpenMP QUIET COMPONENTS CXX)
if(OpenMP_CXX_FOUND)
  target_sources(strata-ik2 PRIVATE ik2_cpu_deep.cpp)
  set_source_files_properties(ik2_cpu_deep.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c;-ffp-contract=off")
  target_compile_definitions(strata-ik2 PRIVATE IK2_HOST_AVX2=1)
  target_link_libraries(strata-ik2 PRIVATE OpenMP::OpenMP_CXX)
endif()

if(STRATA_BUILD_TESTS)
  add_executable(ik2_kernels_test ../../tests/ik2_kernels_test.cu)
  target_link_libraries(ik2_kernels_test PRIVATE ik2_kernels)
  if(STRATA_ENABLE_HIP)
    # No relocatable device code: use the host C++ linker, including its standard runtime libraries.
    set_property(TARGET ik2_kernels_test PROPERTY LINKER_LANGUAGE CXX)
  endif()
  add_test(NAME ik2_kernels_test COMMAND ik2_kernels_test)
  set_tests_properties(ik2_kernels_test PROPERTIES TIMEOUT 300 LABELS gpu)
  if(STRATA_ENABLE_HIP)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME ik2_hip_cli_test COMMAND ${Python3_EXECUTABLE}
      ${PROJECT_SOURCE_DIR}/tests/ik2_hip_cli_test.py $<TARGET_FILE:strata-ik2>)
  endif()
endif()
